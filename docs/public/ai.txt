description: |
  jsh is a Unix shell that serves as a drop-in replacement for Bash and ZSH.
  It executes existing shell scripts unchanged while offering modern syntax
  extensions inspired by Rust, including match expressions, try/catch error
  handling, and named function parameters. It also supports Fish shell syntax
  and maintains full POSIX compliance.

compatibility_matrix:
  bash:
    level: full
    features:
      - if/elif/else/fi conditionals
      - for/while/until loops
      - case statements
      - function definitions
      - pipes and redirections
      - command substitution
      - variable expansion (all forms)
      - arrays (indexed and associative)
      - arithmetic expansion
      - glob expansion
      - background jobs
      - here documents
      
  zsh:
    level: full
    features:
      - all bash features
      - prompt escape sequences (%n, %m, %~, etc.)
      - color sequences (%F{color}...%f)
      - conditional prompts
      - oh-my-zsh theme compatibility
      
  fish:
    level: partial
    supported:
      - set command (-x, -e, -q, -g flags)
      - string builtin (length, upper, lower, split, join, replace, match)
      - math builtin
      - contains builtin
      - status builtin
      - functions builtin
      - abbr (abbreviations)
      - switch/case/end
      - begin/end blocks
      - and/or/not keywords
    not_supported:
      - fish's unique function syntax (function name --argument-names)
      - fish autoloading
      - fish completions system
      
  posix:
    level: full
    features:
      - sh/dash/ash compatible
      - set -e/-u/-x/-n options
      - set -o option names
      - test/[ command
      - . (dot) command
      - trap command
      - getopts
      - readonly
      - local

unique_features:
  match_expression:
    description: "Pattern matching inspired by Rust"
    syntax: |
      match $variable {
          pattern1 => command1
          pattern2 | pattern3 => command2
          4..10 => echo "range match"
          /^regex/ => echo "regex match"
          * => default_command
      }
    patterns:
      - literal: "exact value match"
      - alternation: "pattern1 | pattern2"
      - range: "start..end (numeric)"
      - regex: "/pattern/ (POSIX regex)"
      - wildcard: "* (default/catch-all)"
      
  try_catch_finally:
    description: "Structured error handling"
    syntax: |
      try {
          risky_command
      } catch err {
          echo "Error: $err"
      } finally {
          cleanup_resources
      }
    notes:
      - catch variable contains error message
      - finally always runs
      - replaces complex trap patterns
      
  named_parameters:
    description: "Named function parameters alongside positional"
    syntax: |
      greet(name, greeting) {
          echo "$greeting, $name!"
          echo "Also: $1, $2"
      }
    notes:
      - parameters accessible as named variables
      - also accessible as $1, $2, $3...
      - works with all function syntaxes
      
  let_const:
    description: "Variable bindings with mutability control"
    syntax: |
      let mutable = "can change"
      const IMMUTABLE = "cannot change"
      let mutable = "new value"  # OK
      # const IMMUTABLE = "error"  # Error
      
  loop:
    description: "Infinite loop with break/continue"
    syntax: |
      loop {
          read input
          [ "$input" = "quit" ] && break
      }
      
  fn:
    description: "Function definition shorthand"
    syntax: |
      fn greet { echo "Hello, $1!"; }

    answer: "Yes, jsh is fully compatible with Bash and ZSH scripts. No modifications needed."
    
  - question: "How do I switch from bash to jsh?"
    answer: "Install jsh, add it to /etc/shells, then run 'chsh -s /usr/bin/jsh'."
    
  - question: "Can I use oh-my-zsh themes?"
    answer: "Yes, jsh supports oh-my-zsh prompt escape sequences and git_prompt_info."
    
  - question: "What's the advantage over bash?"
    answer: "Modern syntax (match, try/catch), named parameters, built-in git commands, while maintaining full compatibility."
    
  - question: "Does jsh work on macOS?"
    answer: "Yes, jsh works on macOS including Apple Silicon (M1/M2/M3/M4)."
    
  - question: "How do I see the git branch in my prompt?"
    answer: "Use git_prompt builtin: PS1='\\w$(git_prompt)$ ' or git_branch for just the name."
    
  - question: "What's the difference between .jshrc and .jshenv?"
    answer: ".jshenv is sourced for ALL shells including non-interactive scripts. .jshrc is only for interactive shells."
    
  - question: "Can I mix traditional and modern syntax?"
    answer: "Yes, you can use match alongside case, try/catch alongside trap. Modern syntax is additive."
